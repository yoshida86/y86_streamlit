func crossover_and(ind1,ind2):
    ret_edge = [0 for i in range(頂点数)]
    ret_vertex = []

Input: T:森グラフ P:T上のパス v:P上の最も深さが深い点
Output T:冗長な点と辺を消去した森グラフ
func prunegraph(T,P,v):
    if(PがnewTに加えたパス):
        T = T - P
    else:
        while(vがターミナルでない and vの子がない):
            T = T - v
            v = (vの親ノード)

    return T
